
# 📘 路由编写规则手册（终极简化版）

> **核心理念：极简语法，函数式思维，零冗余符号**

## 🎯 基本结构

### 最简模板

```javascript
// 配置对象
const config = {
  method: 'GET'
};

// 处理函数（函数名必须与文件名相同）
function hello(ctx) {
  ctx.body = {
    message: 'Hello World'
  };
}
```

**就这么简单！** 不需要 `exports`，不需要 `async`，加载器会自动处理！

## ⚙️ config 配置

### 必需字段

```javascript
const config = {
  method: 'GET'  // 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE' | 'ALL'
};
```

### 完整配置

```javascript
const config = {
  // ========== 必需 ==========
  method: 'POST',
  
  // ========== 权限控制 ==========
  requireRoles: ['admin', 'user'],
  requirePermissions: ['posts.create'],
  
  // ========== 事务控制 ==========
  transaction: true,  // 默认 true
  
  // ========== 限流 ==========
  rateLimit: {
    max: 100,
    windowMs: 60000
  },
  
  // ========== 缓存 ==========
  cache: {
    enabled: true,
    ttl: 300
  },
  
  // ========== 审计 ==========
  audit: true,
  auditAction: 'CREATE_POST',
  
  // ========== 数据脱敏 ==========
  sensitiveFields: {
    email: ['admin'],
    phone: ['admin', 'manager']
  },
  
  // ========== 其他 ==========
  description: 'API 描述',
  deprecated: false
};
```

## 📁 目录结构

```
router/
├── private/           # 私有（127.0.0.1）
├── public/            # 公开（无需 JWT）
└── protected/         # 受保护（需要 JWT）
    ├── root/          # 根域名：localhost:3000
    ├── api/           # 子域名：api.localhost:3000
    └── admin:3001/    # 自定义端口：admin.localhost:3001
```

### 路径映射

| 文件路径 | 访问地址 |
|---------|---------|
| `public/root/login.js` | `POST http://localhost:3000/login` |
| `public/api/status.js` | `GET http://api.localhost:3000/status` |
| `protected/api/users/list.js` | `GET http://api.localhost:3000/users/list` |

```javascript
  method: 'POST',
  transaction: false,

    max: 5,
    windowMs: 900000  // 15分钟
  },
  audit: true,
  auditAction: 'USER_LOGIN'
};

function login(ctx) {
  const { username, password } = ctx.request.body;
  
  // 验证用户

    { id: user.id, username: user.username },
    process.env.JWT_SECRET,
    { expiresIn: '24h' }
  );
  
  ctx.body = { token, user };
}
```

### 示例 4：受保护的 API

```javascript
// protected/api/users/profile.js

const config = {
  method: 'GET',
  requireRoles: ['user', 'admin'],
  sensitiveFields: {
    email: ['admin'],
    phone: ['admin']
  }
};

function profile(ctx) {
  const userId = ctx.user.id;  // 来自 JWT
  
  // 使用事务（自动管理）
  const [user] = ctx.transaction.query(
    'SELECT * FROM users WHERE id = ?',
    [userId]
  );
  
  if (!user) {
    ctx.throw(404, '用户不存在');
  }
  
  return user;  // 自动设置 ctx.body，并由中间件脱敏
}
```

### 示例 5：创建资源（201 状态码）

```javascript
// protected/api/posts/create.js

const config = {
  method: 'POST',
  requireRoles: ['user'],
  requirePermissions: ['posts.create'],
  audit: true,
  auditAction: 'CREATE_POST'
};

function create(ctx) {
  const { title, content } = ctx.request.body;
  const userId = ctx.user.id;
  
  // 插入数据库
  const result = ctx.transaction.query(
    'INSERT INTO posts (user_id, title, content) VALUES (?, ?, ?)',
    [userId, title, content]
  );
  
  const postId = result.insertId;
  
  // 记录审计
  ctx.state.auditChanges = { postId, title };
  
  // 设置状态码
  ctx.status = 201;
  ctx.body = { postId, title, content };
}
```

### 示例 6：更新资源

```javascript
// protected/api/posts/update.js

const config = {
  method: 'PUT',
  requirePermissions: ['posts.update'],
  audit: true
};

function update(ctx) {
  const { id, title, content } = ctx.request.body;
  
  // 检查权限
  const [post] = ctx.transaction.query(
    'SELECT * FROM posts WHERE id = ? AND user_id = ?',
    [id, ctx.user.id]
  );
  
  if (!post) {
    ctx.throw(403, '无权修改此文章');
  }
  
  // 更新
  ctx.transaction.query(
    'UPDATE posts SET title = ?, content = ? WHERE id = ?',
    [title, content, id]
  );
  
  return { success: true };
}
```

```javascript
  method: 'DELETE',

  audit: true
};

function delete(ctx) {
  const { id } = ctx.request.body;
  
  // 检查是否存在
  const [post] = ctx.transaction.query(
    'SELECT * FROM posts WHERE id = ? AND user_id = ?',
    [id, ctx.user.id]

    ctx.throw(404, '文章不存在');
  }
  
  // 删除
  ctx.transaction.query('DELETE FROM posts WHERE id = ?', [id]);
  
  // 204 No Content
  ctx.status = 204;
}
```

## 🚀 ctx 对象速查

```javascript
function example(ctx) {
  // ========== 请求 ==========
  ctx.method                // 'GET', 'POST'...
  ctx.url                   // '/users/123?page=1'
  ctx.path                  // '/users/123'
  ctx.query                 // { page: '1' }
  ctx.params                // { id: '123' }
  ctx.request.body          // POST 请求体
  ctx.headers               // 请求头
  ctx.ip                    // 客户端 IP
  
  // ========== 认证 ==========
  ctx.user                  // JWT 用户信息
  ctx.user.id               // 用户 ID
  ctx.user.roles            // 角色数组
  
  // ========== 中间件注入 ==========
  ctx.transaction           // 数据库事务
  ctx.state.routeConfig     // 路由配置
  
  // ========== 响应 ==========
  ctx.status = 201          // 状态码
  ctx.body = { data }       // 响应体
  ctx.set('X-Header', 'v')  // 响应头
  
  // ========== 工具 ==========
  ctx.throw(404, '错误')    // 抛出错误
}
```

## ⚠️ 注意事项

### ✅ 正确

```javascript
// 1. 函数名与文件名一致
// 文件：hello.js
function hello(ctx) { }

// 2. 必须有 config
const config = { method: 'GET' };

// 3. 可以返回值
function api() {
  return { data: 'value' };
}

// 4. 也可以设置 ctx.body
function api(ctx) {
  ctx.body = { data: 'value' };
}
```

### ❌ 错误

```javascript
// 1. 函数名不匹配
// 文件：hello.js
function sayHello(ctx) { }  // ❌ 错误

// 2. 缺少 config
function hello(ctx) { }  // ❌ 错误

// 3. config 缺少 method
const config = {};  // ❌ 错误
function hello(ctx) { }
```
