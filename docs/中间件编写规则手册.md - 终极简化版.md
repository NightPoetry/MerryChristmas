
# 🔧 中间件编写规则手册（终极简化版）

> **核心理念：极简语法，零 exports，函数式思维**

---

## 📐 基本结构

### 最简模板

```javascript
// middleware/logger.js

// 配置
const config = {
  level: ['global'],
  order: 1
};

// 处理函数
function handler(ctx) {
  console.log(`${ctx.method} ${ctx.url}`);
}
```

**就这么简单！** 不需要 `module.exports`，不需要 `async`！

## 🎭 三种中间件模式

### 1. 简单模式

**适用：** 单向处理

```javascript
// middleware/logger.js

const config = {
  level: ['global'],
  order: 1
};

function handler(ctx) {
  console.log(`${ctx.method} ${ctx.url}`);
}
```

---

### 2. 洋葱模式

**适用：** 请求前后都要处理

```javascript
// middleware/transaction.js

const db = require('../db');

const config = {
  level: ['protected'],
  order: 3
};

function before(ctx) {
  ctx.transaction = db.beginTransaction();
  console.log('→ 事务开启');
}

function after(ctx) {
  if (ctx.status < 400) {
    ctx.transaction.commit();
    console.log('✓ 事务提交');
  } else {
    ctx.transaction.rollback();
    console.log('✗ 事务回滚');
  }
}
```

---

### 3. 生命周期模式

**适用：** 完整控制请求周期

```javascript
// middleware/performance.js

const config = {
  level: ['global'],
  order: 1
};

function onRequest(ctx) {
  ctx.state.startTime = Date.now();
}

function before(ctx) {
  ctx.state.beforeTime = Date.now();
}

function after(ctx) {
  ctx.state.afterTime = Date.now();
}

function onResponse(ctx) {
  const duration = Date.now() - ctx.state.startTime;
  ctx.set('X-Response-Time', `${duration}ms`);
}

function onError(ctx, error) {
  console.error('错误:', error.message);
}

function onFinish(ctx) {
  // 异步发送监控数据
  sendMetrics(ctx.state);
}
```

## ⚙️ config 配置

### 必需字段

```javascript
const config = {
  level: ['global'],  // 必需
  order: 10           // 必需
};
```

### level 可选值

| 值 | 说明 |
|----|------|
| `'global'` | 所有路由 |
| `'public'` | 公开路由 |
| `'protected'` | 受保护路由 |
| `'private'` | 私有路由 |

### order 推荐值

| 中间件 | order |
|--------|-------|
| 日志 | 1 |
| CORS | 10 |
| 限流 | 20 |
| JWT | 40 |
| 权限 | 50 |
| 事务 | 60 |
| 格式化 | -2 |
| 错误处理 | -1 |

### 完整配置

```javascript
const config = {
  level: ['global'],
  order: 10,
  enabled: true,              // 可选
  exclude: ['/health'],       // 可选
  description: '日志中间件'   // 可选
};
```

## 📦 完整示例

### 示例 1：日志记录

```javascript
// middleware/logger.js

const config = {
  level: ['global'],
  order: 1
};

function handler(ctx) {
  const start = Date.now();
  console.log(`→ ${ctx.method} ${ctx.url}`);
  
  // 响应后记录
  ctx.res.on('finish', () => {
    const duration = Date.now() - start;
    console.log(`← ${ctx.method} ${ctx.url} ${ctx.status} ${duration}ms`);
  });
}
```

### 示例 2：JWT 验证

```javascript
// middleware/jwtAuth.js

const jwt = require('jsonwebtoken');

const config = {
  level: ['protected'],
  order: 40,
  exclude: ['/auth/refresh']
};

function handler(ctx) {
  const token = ctx.get('Authorization')?.replace('Bearer ', '');
  
  if (!token) {
    ctx.throw(401, '缺少令牌');
  }
  
  try {
    ctx.user = jwt.verify(token, process.env.JWT_SECRET);
  } catch (error) {
    ctx.throw(401, '无效令牌');
  }
}
```

### 示例 3：数据库事务

```javascript
// middleware/transaction.js

const db = require('../db');

const config = {
  level: ['protected'],
  order: 60,
  exclude: ['/health', '/metrics']
};

function before(ctx) {
  const routeConfig = ctx.state.routeConfig || {};
  
  if (routeConfig.transaction === false) {
    return;
  }
  
  ctx.transaction = db.beginTransaction();
  ctx.state.transactionActive = true;
}

function after(ctx) {
  if (!ctx.state.transactionActive) return;
  
  if (ctx.status >= 200 && ctx.status < 400) {
    ctx.transaction.commit();
  } else {
    ctx.transaction.rollback();
  }
}

function onError(ctx, error) {
  if (ctx.state.transactionActive) {
    ctx.transaction.rollback();
  }
}
```

### 示例 4：权限验证

```javascript
// middleware/authorization.js

const config = {
  level: ['protected'],
  order: 50
};

function before(ctx) {
  const routeConfig = ctx.state.routeConfig || {};
  
  // 检查角色
  if (routeConfig.requireRoles) {
    const userRoles = ctx.user.roles || [];
    const hasRole = routeConfig.requireRoles.some(role =>
      userRoles.includes(role)
    );
    
    if (!hasRole) {
      ctx.throw(403, '权限不足');
    }
  }
}

function after(ctx) {
  const routeConfig = ctx.state.routeConfig || {};
  
  // 数据脱敏
  if (ctx.body && routeConfig.sensitiveFields) {
    ctx.body = maskSensitiveData(
      ctx.body,
      routeConfig.sensitiveFields,
      ctx.user.roles
    );
  }
}

function maskSensitiveData(data, fields, roles) {
  // 实现脱敏逻辑
  const result = { ...data };
  
  for (const [field, allowedRoles] of Object.entries(fields)) {
    if (!roles.some(role => allowedRoles.includes(role))) {
      if (result[field]) {
        result[field] = '***';
      }
    }
  }
  
  return result;
}
```

### 示例 5：错误处理

```javascript
// middleware/errorHandler.js

const config = {
  level: ['global'],
  order: -1
};

function onError(ctx, error) {
  console.error('错误详情:', {
    message: error.message,
    stack: error.stack,
    url: ctx.url,
    method: ctx.method
  });
  
  ctx.status = error.status || 500;
  ctx.body = {
    success: false,
    error: {
      code: error.code || 'INTERNAL_ERROR',
      message: error.message
    }
  };
}
```

```javascript
const config = {
  level: ['global'],
  order: -2
};

function after(ctx) {
  // 跳过已格式化的
  if (!ctx.body || ctx.body.success !== undefined) {
    return;
  }
  
  // 统一包装
  ctx.body = {
    success: true,
    data: ctx.body,
    timestamp: Date.now()
  };
}
```

### 示例 7：限流

```javascript
// middleware/rateLimit.js

const rateLimit = require('koa-ratelimit');
const Redis = require('ioredis');

const config = {
  level: ['public', 'protected'],
  order: 20
};

const redis = new Redis();

function handler(ctx) {
  const routeConfig = ctx.state.routeConfig || {};
  
  if (!routeConfig.rateLimit) {
    return;
  }
  
  const { max, windowMs } = routeConfig.rateLimit;
  
  return rateLimit({
    driver: 'redis',
    db: redis,
    duration: windowMs,
    max,
    id: (ctx) => ctx.user?.id || ctx.ip
  })(ctx);
}
```

### 示例 8：CORS

```javascript
// middleware/cors.js

const cors = require('@koa/cors');

const config = {
  level: ['global'],
  order: 10
};

function handler(ctx) {
  return cors({
    origin: (ctx) => {
      const origin = ctx.get('Origin');
      const allowedOrigins = [
        'http://localhost:3000',
        'https://example.com'
      ];
      return allowedOrigins.includes(origin) ? origin : allowedOrigins[0];
    },
    credentials: true,
    allowMethods: ['GET', 'POST', 'PUT', 'DELETE', 'PATCH'],
    allowHeaders: ['Content-Type', 'Authorization']
  })(ctx);
}
```

### 示例 9：审计日志

```javascript
// middleware/audit.js

const config = {
  level: ['protected'],
  order: 2
};

function before(ctx) {
  ctx.state.audit = {
    timestamp: new Date(),
    userId: ctx.user?.id,
    ip: ctx.ip,
    method: ctx.method,
    url: ctx.url,
    body: sanitizeBody(ctx.request.body)
  };
}

function after(ctx) {
  const routeConfig = ctx.state.routeConfig || {};
  
  if (!routeConfig.audit) return;
  
  ctx.state.audit.status = ctx.status;
  ctx.state.audit.success = ctx.status < 400;
  ctx.state.audit.duration = Date.now() - ctx.state.audit.timestamp.getTime();
  
  if (routeConfig.auditAction) {
    ctx.state.audit.action = routeConfig.auditAction;
  }
}

function onFinish(ctx) {
  if (ctx.state.audit) {
    saveAuditLog(ctx.state.audit);
  }
}

function sanitizeBody(body) {
  // 移除敏感信息
  const sanitized = { ...body };
  delete sanitized.password;
  delete sanitized.token;
  return sanitized;
}
```

### 示例 10：性能监控

```javascript
// middleware/performance.js

const config = {
  level: ['global'],
  order: 1
};

function onRequest(ctx) {
  ctx.state.perf = {
    start: Date.now(),
    marks: {}
  };
}

function before(ctx) {
  ctx.state.perf.marks.beforeBusiness = Date.now();
}

function after(ctx) {
  ctx.state.perf.marks.afterBusiness = Date.now();
}

function onResponse(ctx) {
  const perf = ctx.state.perf;
  const total = Date.now() - perf.start;
  const business = perf.marks.afterBusiness - perf.marks.beforeBusiness;
  
  ctx.set('X-Response-Time', `${total}ms`);
  ctx.set('X-Business-Time', `${business}ms`);
  
  if (total > 1000) {
    console.warn(`⚠ 慢请求: ${ctx.method} ${ctx.url} - ${total}ms`);
  }
}

function onFinish(ctx) {
  // 发送到监控系统
  sendMetrics({
    url: ctx.url,
    method: ctx.method,
    duration: Date.now() - ctx.state.perf.start,
    status: ctx.status
  });
}

async function sendMetrics(data) {
  // 异步发送
}
```

## 🎯 执行流程

```
HTTP 请求
    ↓
[logger.handler]           order: 1
[cors.handler]             order: 10
[jwtAuth.handler]          order: 40
[authorization.before]     order: 50
[transaction.before]       order: 60
    ↓
【业务路由】
    ↓
[transaction.after]        order: 60
[authorization.after]      order: 50
[responseFormatter.after]  order: -2
[errorHandler.onError]     order: -1
    ↓
HTTP 响应
    ↓
[audit.onFinish]          异步
[performance.onFinish]    异步
```
